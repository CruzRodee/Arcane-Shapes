SetLayerMedia(background -m Classroom)
CreateCharacter(Oz)
Oz.SetPosition(-x 1), Oz.Show()
PlaySong(Upbeat)

SetLayerMedia(background -m Intro_Circle1)

SaveCheckpoint(LO_Intro_Semicircle_START)

$formulaQuestion = ""
$question1 = 0
$question2 = 0

Oz [talk] "Ngayon naman, SEMI-CIRCLE tayo!"

"Ano ba ang SEMI? Ibigsabihin nito ay HALF! KALAHATI!"

SetLayerMedia(background -m Intro_SemiCircle1)

"Kaya ang SEMI-CIRCLE ay HALF-CIRCLE!"

"Kaya pag sinabi na SEMI-CIRCLE. . .{c} ito ay CIRCLE pero kalahati lang."

SetLayerMedia(background -m Intro_SemiCircle2)

"Ibig sabihin, kung kaya mong kunin ang area ng isang buong CIRCLE, madali na rin ito!"

SetLayerMedia(background -m Intro_SemiCircle4)

"Ang formula ng CIRCLE ay r x r x 3.14."
"Para sa SEMI-CIRCLE, gagawin mo lang itong kalahati — kaya hatiin sa 2!"
"Kaya ang formula ay r x r x 3.14 ÷ 2."

"Ang r ay RADIUS, tulad ng dati — ang sukat mula sa gitna hanggang sa gilid ng bilog."

choice "Ano ang FORMULA ng SEMI-CIRCLE?"
{
    - r x r x 3.14 ÷ 2
        Oz [surprised] "Tama! Dahil kalahati lang ng CIRCLE, hatiin lang natin sa 2."
        $formulaQuestion = "rxrx3.14div2"
    - r x r x 3.14
        Oz [sighsad] "Mali. Iyan ay para sa buong CIRCLE."
        $formulaQuestion = "rxrx3.14"
    - L x W
        Oz [sighsad] "Mali. Para iyon sa RECTANGLE."
        $formulaQuestion = "LxW"
    - B x H ÷ 2
        Oz [sighsad] "Mali. Para iyon sa TRIANGLE."
        $formulaQuestion = "BxHdiv2"
}

Oz [talk] "Subukan natin gamitin ang formula!"

input "Kung r = 4, ano ang AREA ng SEMI-CIRCLE? (NUMERO LAMANG, gamit ang 3.14)"
$ans1 = "<input>"
$question1 = $ans1

if ($ans1 == 25.12)
{
    Oz [surprised] "Tama! (4 x 4 x 3.14) ÷ 2 = 25.12."
}
else
{
    Oz [sighsad] "Hindi pa tama. Tandaan: (r x r x 3.14) ÷ 2."
    "(4 x 4 x 3.14) ÷ 2 = 25.12."
}

input "Isa pa. Kung r = 2, ano ang AREA? (NUMERO LAMANG, gamit ang 3.14)"
$ans2 = "<input>"
$question2 = $ans2

if ($ans2 == 6.28)
{
    Oz [surprised] "Magaling! (2 x 2 x 3.14) ÷ 2 = 6.28."
}
else
{
    Oz [sighsad] "Mali. Ulitin natin: (2 x 2 x 3.14) ÷ 2 = 6.28."
}

Oz [talk] "Sige! Tandaan: ang SEMI-CIRCLE ay kalahati lang ng CIRCLE — kaya hatiin lang sa 2!"

SaveCheckpoint(LO_Intro_Semicircle_END formulaQuestion:<$formulaQuestion> question1:<$question1> question2:<$question2>)

StopSong(), StopAmbience()
